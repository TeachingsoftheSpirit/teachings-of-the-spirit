'use client'

import { useEffect, useState } from 'react'
import Link from 'next/link'

type SavedItem = {
  id: string
  teaching_number: number
  memo: string | null
  teachings: { title: string; date: string } | null
}

const GARAMOND =
  '"EB Garamond", Garamond, "Palatino Linotype", Palatino, "Times New Roman", serif'

export default function CheckMarginalia() {
  const [open, setOpen] = useState(false)
  const [loading, setLoading] = useState(false)
  const [items, setItems] = useState<SavedItem[]>([])
  const [error, setError] = useState('')

  useEffect(() => {
    if (!open) return
    let cancelled = false
    const load = async () => {
      setLoading(true)
      setError('')
      try {
        const res = await fetch('/api/saved-teachings')
        if (!res.ok) {
          if (res.status === 401) {
            setError('Sign in to see your Marginalia.')
            setItems([])
            return
          }
          throw new Error('Could not load Marginalia')
        }
        const data = await res.json()
        const all: SavedItem[] = data.saved || []
        if (!cancelled) {
          setItems(all.filter((s) => s.memo && s.memo.trim()))
        }
      } catch (err: any) {
        if (!cancelled) setError(err.message || 'Could not load')
      } finally {
        if (!cancelled) setLoading(false)
      }
    }
    load()
    return () => {
      cancelled = true
    }
  }, [open])

  return (
    <>
      <button
        type="button"
        onClick={() => setOpen(true)}
        className="text-[13px] text-[#6B5E54] hover:text-[#2C2522] underline underline-offset-2 transition-colors"
        style={{ fontFamily: GARAMOND }}
      >
        Check Marginalia
      </button>

      {open && (
        <div className="fixed inset-0 z-[70] flex items-center justify-center p-4">
          <div
            className="absolute inset-0 bg-black/30"
            onClick={() => setOpen(false)}
          />
          <div
            className="relative w-full max-w-md max-h-[min(480px,80vh)] flex flex-col rounded-sm shadow-2xl overflow-hidden"
            style={{
              background: '#F7F1E6',
              border: '1px solid #8A735A',
            }}
          >
            <div
              className="flex items-center justify-between px-4 py-3 shrink-0"
              style={{
                background:
                  'linear-gradient(180deg, #5c4a3a 0%, #3d342f 100%)',
              }}
            >
              <div>
                <div
                  className="text-[15px] text-[#F5F0E6] font-medium"
                  style={{ fontFamily: GARAMOND }}
                >
                  Marginalia
                </div>
                <div className="text-[11px] text-[#C9BDA8]">
                  Your notes in the margins
                </div>
              </div>
              <button
                type="button"
                onClick={() => setOpen(false)}
                className="text-[#F5F0E6] hover:text-white text-sm px-1"
              >
                ✕
              </button>
            </div>

            <div className="flex-1 overflow-y-auto p-4">
              {loading ? (
                <p className="text-[13px] text-[#6B5E54]">Loading…</p>
              ) : error ? (
                <p className="text-[13px] text-[#7A3E3E]">{error}</p>
              ) : items.length === 0 ? (
                <p
                  className="text-[13px] text-[#6B5E54] leading-relaxed"
                  style={{ fontFamily: GARAMOND }}
                >
                  No Marginalia yet. Open your desk, open Saved Teachings, and
                  leave a note in the margin of any Teaching you have saved.
                </p>
              ) : (
                <ul className="space-y-4">
                  {items.map((item) => (
                    <li
                      key={item.id}
                      className="pb-3 border-b border-[#E5DFD3] last:border-0"
                    >
                      <Link
                        href={`/teachings/${item.teaching_number}`}
                        onClick={() => setOpen(false)}
                        className="text-[14px] font-semibold text-[#2C2522] hover:underline"
                        style={{ fontFamily: GARAMOND }}
                      >
                        {item.teachings?.title ||
                          `Teaching ${item.teaching_number}`}
                      </Link>
                      {item.teachings?.date && (
                        <div className="text-[11px] text-[#6B5E54] mt-0.5">
                          {item.teachings.date}
                        </div>
                      )}
                      <p
                        className="mt-1.5 text-[13px] text-[#3F362E] leading-relaxed italic"
                        style={{ fontFamily: GARAMOND }}
                      >
                        {item.memo}
                      </p>
                    </li>
                  ))}
                </ul>
              )}
            </div>
          </div>
        </div>
      )}
    </>
  )
}
