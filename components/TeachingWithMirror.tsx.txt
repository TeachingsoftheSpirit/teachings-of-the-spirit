'use client'

import { useRef } from 'react'
import GaladrielsMirror, { GaladrielsMirrorHandle } from './GaladrielsMirror'

type Segment = {
  start: number
  end: number
  text: string
}

type Props = {
  videoUrl: string
  title: string
  paragraphs: string[]
  segments: Segment[]
}

export default function TeachingWithMirror({
  videoUrl,
  title,
  paragraphs,
  segments,
}: Props) {
  const mirrorRef = useRef<GaladrielsMirrorHandle>(null)

  // Find the best matching start time for a paragraph
  function findStartTime(para: string): number | null {
    const clean = para.trim().toLowerCase().slice(0, 60)
    if (!clean) return null

    for (const seg of segments) {
      if (seg.text.toLowerCase().includes(clean.slice(0, 40))) {
        return seg.start
      }
    }

    // Fallback: first few words match
    const words = clean.split(/\s+/).slice(0, 6).join(' ')
    for (const seg of segments) {
      if (seg.text.toLowerCase().includes(words)) {
        return seg.start
      }
    }

    return null
  }

  function handleParagraphClick(para: string) {
    const start = findStartTime(para)
    if (start !== null) {
      mirrorRef.current?.seekTo(start)
    } else {
      // Just open the mirror if no match
      mirrorRef.current?.open()
    }
  }

  return (
    <>
      <div className="text-center mb-8">
        <GaladrielsMirror
          ref={mirrorRef}
          videoUrl={videoUrl}
          title={title}
        />
      </div>

      <article>
        <div className="text-[#2C2522] leading-[1.85] text-[1.12rem] space-y-5">
          {paragraphs.map((para, i) => {
            const hasMatch = findStartTime(para) !== null
            return (
              <p
                key={i}
                onClick={() => handleParagraphClick(para)}
                className={
                  hasMatch
                    ? 'cursor-pointer hover:bg-[#EDE6DC] rounded px-1 -mx-1 transition-colors'
                    : ''
                }
                title={hasMatch ? 'Click to hear this moment in Galadriel’s Mirror' : undefined}
              >
                {para}
              </p>
            )
          })}
        </div>
      </article>
    </>
  )
}
